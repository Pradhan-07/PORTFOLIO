#include<iostream>
using namespace std;

class Node{ // User Defined Data Type
public:
    int val;
    Node* next;
    Node(int val){
        this->val = val;
    }
};

class MyLinkedList{ // User Defined Data Type
private:
    Node* head;
    Node* tail;
    int length;
public:
    MyLinkedList(){
        head = tail = NULL;
        length = 0;
    }

    void display(){
        Node* temp = head;
        while(temp != NULL){
            cout<<temp->val<<" ";
            temp = temp->next;
        }
        cout<<endl;
    }

    void insertAtTail(int val){
        Node* n = new Node(val);
        if(length == 0) head = tail = n;
        else{
            tail->next = n;
            tail = n;
        }
        length++;
    }

    void insertAtHead(int val){
        Node* n = new Node(val);
        if(length == 0) head = tail = n;
        else{
            n->next = head;
            head = n;
        }
        length++;
    }

    void insert(int idx, int val){
        if(idx<0 or idx>length){
            cout<<"Invalid Index!"<<endl;
            return;
        }
        if(idx==0){ // insert at head
            insertAtHead(val);
            return;
        }
        if(idx==length){ // insert at tail
            insertAtTail(val);
            return;
        }
        // Actual Code
        Node* n = new Node(val);
        Node* temp = head;
        for(int i=1;i<=idx-1;i++){
            temp = temp->next;
        }
        n->next = temp->next;
        temp->next = n;
        length++;
    }

    void removeAtHead(){
        if(length == 0){
            cout<<"List is Empty!"<<endl;
            return;
        }
        Node* toBeDeleted = head;
        head = head->next;
        delete toBeDeleted;
        length--;
    }

    void remove(int idx){
        if(idx<0 or idx>=length){
            cout<<"Invalid Index"<<endl;
            return;
        }
        if(idx==0){
            removeAtHead();
            return;
        }
        Node* temp = head;
        for(int i=1;i<=idx-1;i++){
            temp = temp->next;
        }
        Node* toBeDeleted = temp->next;
        temp->next = temp->next->next;
        if(idx == length-1) tail = temp; // IMPORTANT
        delete toBeDeleted;
        length--;
    }

    int get(int idx){
        if(idx<0 or idx>=length){
            cout<<"Invalid Index!"<<endl;
            return -1;
        }
        Node* temp = head;
        for(int i=1;i<=idx;i++){
            temp = temp->next;
        }
        return temp->val;
    }

    int size(){
        return length;
    }
};

int main(){

    list.display();

    list.display();

}

    int i =0,j=0,k=0;
    while(i<a.size() && j<b.size()){
        if(a[i]>b[j]) c[k++]=b[j++];
        else c[k++]=a[i++];
    }

}

    if(n<=1) return;

    vector<int> b(n-n/2);
    int idx =0;

        a[i] = arr[idx++];
    }
    for(int i =0; i<n-n/2;i++){
        b[i] = arr[idx++];
    }
    mergesort(a);

}

// ===== PORTFOLIO — src/components/CanvasWall.tsx (tsx) =====
// https://github.com/Pradhan-07/PORTFOLIO/blob/42da8fbf53dc926a1501a9ce208f45f88febe733/src/components/CanvasWall.tsx
import React, { useEffect, useRef } from 'react';
import type { Theme } from '../types';
import { CODE_WALL_SNIPPETS } from '../data/codeSnippets';
import { usePrefersReducedMotion } from '../hooks/usePrefersReducedMotion';

interface CanvasWallProps {
  theme: Theme;
  interactive?: boolean;
}

export const CanvasWall: React.FC<CanvasWallProps> = ({ theme, interactive = true }) => {
  const reducedMotion = usePrefersReducedMotion();
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const mouseRef = useRef<{ x: number; y: number; active: boolean }>({ x: -100, y: -100, active: false });

  useEffect(() => {
    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d', { alpha: false });
    if (!ctx) return;

    let animId: number;
    let drawFrame: () => void = () => undefined;
    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = width * dpr;
    canvas.height = height * dpr;
    ctx.scale(dpr, dpr);

    const CELL_W = 7.5;
    const CELL_H = 14;
    const FONT_SIZE = 11;
    const FONT_FAMILY = '"IBM Plex Mono", ui-monospace, monospace';

    const cols = Math.ceil(width / CELL_W) + 2;
    const rows = Math.ceil(height / CELL_H) + 2;

    // Generate repeated code matrix grid
    const codeLines = CODE_WALL_SNIPPETS;
    const grid: {
      char: string;
      kind: 'keyword' | 'string' | 'number' | 'function' | 'type' | 'comment' | 'text' | 'member';
      baseAlpha: number;
    }[][] = [];

    let lineIdx = 0;
    for (let r = 0; r < rows; r++) {
      const lineData = codeLines[lineIdx % codeLines.length];
      lineIdx++;

      const rowChars: {
        char: string;
        kind: 'keyword' | 'string' | 'number' | 'function' | 'type' | 'comment' | 'text' | 'member';
        baseAlpha: number;
      }[] = [];

      let charCount = 0;
      let tokIdx = 0;

      while (charCount < cols) {
        if (tokIdx >= lineData.tokens.length) {
          // Fill rest of row with subtle space or comment indentation
          rowChars.push({ char: ' ', kind: 'text', baseAlpha: 0.1 });
          charCount++;
        } else {
          const token = lineData.tokens[tokIdx];
          tokIdx++;
          for (let i = 0; i < token.text.length && charCount < cols; i++) {
            const ch = token.text[i];
            const isComment = token.kind === 'comment';
            rowChars.push({
              char: ch,
              kind: token.kind,
              baseAlpha: isComment ? 0.22 : 0.42,
            });
            charCount++;
          }
        }
      }
      grid.push(rowChars);
    }

    // Interactive flux state
    let frame = 0;
    const cursor = { row: 3, col: 12, blink: true };

    const handleMouseMove = (e: MouseEvent) => {
      if (!interactive) return;
      mouseRef.current = { x: e.clientX, y: e.clientY, active: true };
    };

    const handleMouseLeave = () => {
      mouseRef.current.active = false;
    };

    const handleResize = () => {
      if (!canvas) return;
      width = canvas.width = window.innerWidth;
      height = canvas.height = window.innerHeight;
      canvas.width = width * dpr;
      canvas.height = height * dpr;
      ctx.scale(dpr, dpr);
      // In reduced-motion mode there is no loop; keep the static frame fresh.
      if (reducedMotion) drawFrame();
    };

    window.addEventListener('mousemove', handleMouseMove);
    window.addEventListener('mouseleave', handleMouseLeave);
    window.addEventListener('resize', handleResize);

    const render = () => {
      frame++;
      ctx.fillStyle = theme.bg;
      ctx.fillRect(0, 0, width, height);

      ctx.font = `400 ${FONT_SIZE}px ${FONT_FAMILY}`;
      ctx.textBaseline = 'top';

      const mouseX = mouseRef.current.x;
      const mouseY = mouseRef.current.y;
      const mouseActive = mouseRef.current.active;

      // Draw subtle grid characters
      for (let r = 0; r < rows; r++) {
        const row = grid[r];
        if (!row) continue;
        const y = r * CELL_H;

        for (let c = 0; c < cols; c++) {
          const item = row[c];
          if (!item || item.char === ' ') continue;

          const x = c * CELL_W;

          // Distance from mouse for interactive glow
          let alpha = item.baseAlpha;
          if (mouseActive) {
            const dx = x - mouseX;
            const dy = y - mouseY;
            const dist = Math.sqrt(dx * dx + dy * dy);
            if (dist < 140) {
              const boost = (1 - dist / 140) * 0.55;
              alpha = Math.min(1.0, alpha + boost);
            }
          }

          // Subtle rhythmic breathing in background
          if ((r + c + frame) % 180 === 0) {
            alpha = Math.min(0.9, alpha + 0.3);
          }

          // Color selection based on token kind
          let color = theme.kinds[item.kind] || theme.fg;

          ctx.fillStyle = color;
          ctx.globalAlpha = alpha;
          ctx.fillText(item.char, x, y + 1);
        }
      }

      // Blinking terminal cursor at row 3, col 28
      if (frame % 60 < 35) {
        ctx.fillStyle = theme.accent;
        ctx.globalAlpha = 0.85;
        ctx.fillRect(cursor.col * CELL_W, cursor.row * CELL_H + 2, CELL_W, CELL_H - 2);
      }

      ctx.globalAlpha = 1.0;
    };

    drawFrame = render;

    if (reducedMotion) {
      // Calm state (G9): draw a single static frame — no breathing, no
      // blinking cursor loop, no mouse-glow animation. Content is identical.
      render();
    } else {
      const loop = () => {
        render();
        animId = requestAnimationFrame(loop);
      };
      loop();
    }

    return () => {
      cancelAnimationFrame(animId);
      window.removeEventListener('mousemove', handleMouseMove);
      window.removeEventListener('mouseleave', handleMouseLeave);
      window.removeEventListener('resize', handleResize);
    };
  }, [theme, interactive, reducedMotion]);

  return (
    <canvas
      ref={canvasRef}
      className="wall-canvas"
      aria-hidden="true"
      style={{
        position: 'fixed',
        inset: 0,
        width: '100vw',
        height: '100vh',
        zIndex: 0,
        pointerEvents: 'none',
        display: 'block',
      }}
    />
  );
};

// https://github.com/Pradhan-07/PORTFOLIO/blob/42da8fbf53dc926a1501a9ce208f45f88febe733/src/App.tsx
import React, { useState, useEffect, useCallback } from 'react';
import { THEMES } from './data/themes';
import type { Theme } from './types';
import { CanvasWall } from './components/CanvasWall';
import { LandingScreen } from './components/LandingScreen';
import { DashboardPortfolio } from './components/hancore/DashboardPortfolio';
import { RadialEclipseTransition } from './components/hancore/RadialEclipseTransition';
import { usePrefersReducedMotion } from './hooks/usePrefersReducedMotion';

const THEME_STORAGE_KEY = 'karan-portfolio-theme';

const readStoredThemeIndex = (): number => {
  try {
    const raw = window.localStorage.getItem(THEME_STORAGE_KEY);
    const parsed = raw === null ? NaN : Number.parseInt(raw, 10);
    return Number.isInteger(parsed) && parsed >= 0 && parsed < THEMES.length ? parsed : 0;
  } catch {
    return 0;
  }
};

export const App: React.FC = () => {
  const reducedMotion = usePrefersReducedMotion();
  const [themeIndex, setThemeIndex] = useState<number>(readStoredThemeIndex);
  const [screen, setScreen] = useState<'landing' | 'portfolio'>('landing');
  const [isTransitioning, setIsTransitioning] = useState(false);

  const currentTheme: Theme = THEMES[themeIndex];

  // Apply theme variables to root document
  const applyThemeVariables = useCallback((theme: Theme) => {
    const root = document.documentElement;
    root.style.setProperty('--bg', theme.bg);
    root.style.setProperty('--fg', theme.fg);
    root.style.setProperty('--ink', theme.fg);
    root.style.setProperty('--dim', theme.dim);
    root.style.setProperty('--chrome', theme.chrome);
    root.style.setProperty('--accent', theme.accent);
    root.style.setProperty('--line', theme.line);
    root.style.setProperty('--kw', theme.kinds.keyword);
    root.style.setProperty('--str', theme.kinds.string);
    root.style.setProperty('--fn', theme.kinds.function);
    root.style.setProperty('--ty', theme.kinds.type);
    root.style.setProperty('--num', theme.kinds.number);
    root.style.colorScheme = theme.light ? 'light' : 'dark';
  }, []);

  useEffect(() => {
    applyThemeVariables(currentTheme);
  }, [currentTheme, applyThemeVariables]);

  // Persist the chosen theme so it is restored on the next visit.
  useEffect(() => {
    try {
      window.localStorage.setItem(THEME_STORAGE_KEY, String(themeIndex));
    } catch {
      // Storage unavailable (private mode etc.) — theme still applies for the session.
    }
  }, [themeIndex]);

  const handleToggleTheme = () => {
    setThemeIndex((prev) => (prev + 1) % THEMES.length);
  };

  const handleSelectTheme = (index: number) => {
    if (index >= 0 && index < THEMES.length) {
      setThemeIndex(index);
    }
  };

  // Trigger Part 2: Radial Eclipse Transition
  // Reduced motion (G9): no cinematic sequence — go straight to the portfolio.
  const handleStartEnterTransition = () => {
    if (reducedMotion) {
      handleTransitionComplete();
      return;
    }
    setIsTransitioning(true);
  };

  // Called when Radial Eclipse Transition completes (Frames 08 - 12)
  const handleTransitionComplete = () => {
    setScreen('portfolio');
    setIsTransitioning(false);
    window.scrollTo(0, 0);
  };

  const handleBackToLanding = () => {
    setScreen('landing');
    setIsTransitioning(false);
    window.scrollTo(0, 0);
  };

  return (
    <div
      className="app-wrapper"
      style={{
        position: 'relative',
        minHeight: '100vh',
        background: '#080908',
        color: 'var(--fg)',
        overflowX: 'hidden',
      }}
    >
      {/* Background Matrix Canvas Wall */}
      <CanvasWall theme={currentTheme} interactive={screen === 'landing' && !isTransitioning} />

      {/* Screens */}
      {screen === 'landing' ? (
        <LandingScreen
          theme={currentTheme}
          onEnter={handleStartEnterTransition}
          onToggleTheme={handleToggleTheme}
          isLeaving={isTransitioning}
        />
      ) : (
        <DashboardPortfolio
          theme={currentTheme}
          onBackToIntro={handleBackToLanding}
          onToggleTheme={handleToggleTheme}
          onSelectTheme={handleSelectTheme}
        />
      )}

      {/* Part 2: Signature Radial Eclipse Expanding Burst Transition */}
      {isTransitioning && (
        <RadialEclipseTransition
          onComplete={handleTransitionComplete}
          themeAccent={currentTheme.accent}
        />
      )}
    </div>
  );
};

export default App;

// ===== PORTFOLIO — src/components/hancore/RadialEclipseTransition.tsx (tsx) =====
// https://github.com/Pradhan-07/PORTFOLIO/blob/42da8fbf53dc926a1501a9ce208f45f88febe733/src/components/hancore/RadialEclipseTransition.tsx
import React, { useEffect, useRef } from 'react';
import { usePrefersReducedMotion } from '../../hooks/usePrefersReducedMotion';

interface RadialEclipseTransitionProps {
  onComplete: () => void;
  themeAccent?: string;
}

export const RadialEclipseTransition: React.FC<RadialEclipseTransitionProps> = ({
  onComplete,
  themeAccent = '#ebe6db',
}) => {
  const canvasRef = useRef<HTMLCanvasElement | null>(null);
  const reducedMotion = usePrefersReducedMotion();

  useEffect(() => {
    // Reduced motion (G9): skip the 1.85s canvas sequence entirely and
    // hand over to the portfolio almost immediately.
    if (reducedMotion) {
      const t = window.setTimeout(() => onComplete(), 60);
      return () => window.clearTimeout(t);
    }

    const canvas = canvasRef.current;
    if (!canvas) return;

    const ctx = canvas.getContext('2d');
    if (!ctx) return;

    let width = (canvas.width = window.innerWidth);
    let height = (canvas.height = window.innerHeight);

    const dpr = Math.min(window.devicePixelRatio || 1, 2);
    canvas.width = width * dpr;
    canvas.height = height * dpr;
    ctx.scale(dpr, dpr);

    const cx = width / 2;
    const cy = height / 2;

    const DURATION = 1850; // Total transition time in ms (~1.85s)
    const startTime = performance.now();

    // Procedural radial streaks (280 rays of varying length, angle, and thickness)
    const RAY_COUNT = 320;
    const rays = Array.from({ length: RAY_COUNT }, (_, i) => {
      const angle = (i / RAY_COUNT) * Math.PI * 2;
      const lengthMult = 0.5 + Math.random() * 0.9;
      const alphaMult = 0.3 + Math.random() * 0.7;
      const widthMult = 0.6 + Math.random() * 1.2;
      return { angle, lengthMult, alphaMult, widthMult };
    });

    // Orbiting particles in core
    const PARTICLE_COUNT = 60;
    const particles = Array.from({ length: PARTICLE_COUNT }, () => ({
      angle: Math.random() * Math.PI * 2,
      dist: 20 + Math.random() * 50,
      speed: (Math.random() - 0.5) * 0.05,
      size: 1 + Math.random() * 2,
    }));

    let animId: number;

    const render = (now: number) => {
      const elapsed = now - startTime;
      const progress = Math.min(1.0, elapsed / DURATION);

      ctx.clearRect(0, 0, width, height);

      // Phase calculation
      // 0.0 -> 0.25: Core convergence
      // 0.25 -> 0.60: Ring formation & particle swirl
      // 0.60 -> 0.90: Eclipse burst expansion
      // 0.90 -> 1.00: Fullscreen takeover & resolve

      if (progress < 0.25) {
        // Phase 1: Small particle cluster convergence
        const p1 = progress / 0.25;
        ctx.fillStyle = `color-mix(in srgb, ${themeAccent} ${p1 * 70}%, transparent)`;
        particles.forEach((pt) => {
          const currentDist = pt.dist * (1 - p1 * 0.5);
          const x = cx + Math.cos(pt.angle) * currentDist;
          const y = cy + Math.sin(pt.angle) * currentDist;
          ctx.beginPath();
          ctx.arc(x, y, pt.size, 0, Math.PI * 2);
          ctx.fill();
        });

        // Small center core
        ctx.strokeStyle = `color-mix(in srgb, ${themeAccent} ${p1 * 80}%, transparent)`;
        ctx.lineWidth = 1;
        ctx.beginPath();
        ctx.arc(cx, cy, 14 * p1, 0, Math.PI * 2);
        ctx.stroke();

      } else if (progress < 0.60) {
        // Phase 2: Ring formation & micro-rays
        const p2 = (progress - 0.25) / 0.35;
        const ringRadius = 24 + p2 * 70;

        // Draw radial streaks emerging from ring perimeter
        rays.forEach((ray) => {
          const rInner = ringRadius;
          const rOuter = ringRadius + ray.lengthMult * 60 * p2;

          ctx.strokeStyle = `color-mix(in srgb, ${themeAccent} ${(ray.alphaMult * p2 * 85).toFixed(1)}%, transparent)`;
          ctx.lineWidth = ray.widthMult;
          ctx.beginPath();
          ctx.moveTo(cx + Math.cos(ray.angle) * rInner, cy + Math.sin(ray.angle) * rInner);
          ctx.lineTo(cx + Math.cos(ray.angle) * rOuter, cy + Math.sin(ray.angle) * rOuter);
          ctx.stroke();
        });

        // Luminous perimeter
        ctx.strokeStyle = themeAccent;
        ctx.lineWidth = 1.5;
        ctx.beginPath();
        ctx.arc(cx, cy, ringRadius, 0, Math.PI * 2);
        ctx.stroke();

        // Dark circular center
        ctx.fillStyle = '#050505';
        ctx.beginPath();
        ctx.arc(cx, cy, ringRadius - 1, 0, Math.PI * 2);
        ctx.fill();

        // Technical coordinates text
        ctx.font = '9px "IBM Plex Mono", monospace';
        ctx.fillStyle = `color-mix(in srgb, ${themeAccent} 50%, transparent)`;
        ctx.textAlign = 'center';
        ctx.fillText(`PORTAL // 0x${Math.round(p2 * 255).toString(16).toUpperCase()}`, cx, cy + 3);

      } else {
        // Phase 3 & 4: Full-screen radial expansion & eclipse takeover
        const p3 = (progress - 0.60) / 0.40;
        const maxDiagonal = Math.hypot(width, height) * 1.1;
        const ringRadius = 94 + p3 * maxDiagonal;

        // Massive radial burst
        rays.forEach((ray) => {
          const rInner = ringRadius;
          const rOuter = ringRadius + ray.lengthMult * 180 * (1 - p3 * 0.4);

          ctx.strokeStyle = `color-mix(in srgb, ${themeAccent} ${(ray.alphaMult * (1 - p3 * 0.6) * 90).toFixed(1)}%, transparent)`;
          ctx.lineWidth = ray.widthMult * 1.5;
          ctx.beginPath();
          ctx.moveTo(cx + Math.cos(ray.angle) * rInner, cy + Math.sin(ray.angle) * rInner);
          ctx.lineTo(cx + Math.cos(ray.angle) * rOuter, cy + Math.sin(ray.angle) * rOuter);
          ctx.stroke();
        });

        // Expanding dark circular center that consumes the screen
        ctx.fillStyle = '#050505';
        ctx.beginPath();
        ctx.arc(cx, cy, ringRadius, 0, Math.PI * 2);
        ctx.fill();

        // Luminous ring leading the charge
        if (p3 < 0.95) {
          ctx.strokeStyle = themeAccent;
          ctx.lineWidth = 2 * (1 - p3);
          ctx.beginPath();
          ctx.arc(cx, cy, ringRadius, 0, Math.PI * 2);
          ctx.stroke();
        }
      }

      if (progress < 1.0) {
        animId = requestAnimationFrame(render);
      } else {
        onComplete();
      }
    };

    animId = requestAnimationFrame(render);

    return () => {
      cancelAnimationFrame(animId);
    };
  }, [onComplete, themeAccent, reducedMotion]);

  if (reducedMotion) return null;

  return (
    <canvas
      ref={canvasRef}
      className="radial-eclipse-canvas"
      style={{
        position: 'fixed',
        inset: 0,
        width: '100vw',
        height: '100vh',
        zIndex: 9999,
        pointerEvents: 'none',
        display: 'block',
      }}
    />
  );
};

// ===== study_detection — app.py (python) =====
// https://github.com/Pradhan-07/study_detection/blob/a79829824e27f413179747a491880f33ff63ac96/app.py
import cv2
import cvzone
from cvzone.FaceMeshModule import FaceMeshDetector
from ultralytics import YOLO
import pygame

# Initialize Pygame Mixer
pygame.mixer.init()
alarm_sleep = pygame.mixer.Sound("alarm.mp3")       # Sleep alarm
alarm_facehide = pygame.mixer.Sound("faudio.mp3")   # Face hidden/covered alarm
alarm_phone = pygame.mixer.Sound("paudio.mp3")      # Phone alarm

# Track active audio state ('sleep', 'facehide', 'phone', or None)
current_playing = None

# Initialize Camera, Face Mesh, and YOLO Model
cap = cv2.VideoCapture(0)
face_detector = FaceMeshDetector(maxFaces=1)
phone_detector = YOLO("yolov8n.pt")  # COCO pre-trained lightweight model

# COCO Dataset Class Names
classNames = phone_detector.names  # Class '67' is 'cell phone' in standard COCO

# Landmark Indices for Sleep Detection
LEFT_EYE_TOP = 159
LEFT_EYE_BOTTOM = 145
FACE_LEFT = 130
FACE_RIGHT = 243

# Frame Counters & Thresholds
closed_frames = 0
SLEEP_THRESHOLD_FRAMES = 15

covered_frames = 0
COVER_THRESHOLD_FRAMES = 20  # ~0.6 seconds of missing face triggers warning

while True:
    success, img = cap.read()
    if not success:
        break

    # Check if ANY sound is currently playing to completion
    is_audio_busy = pygame.mixer.get_busy()

    # If audio finished playing, reset tracking state to None
    if not is_audio_busy:
        current_playing = None

    # ------------------- 1. SLEEP & FACE COVER DETECTION -------------------
    img, faces = face_detector.findFaceMesh(img, draw=False)
    is_sleepy = False
    is_face_covered = False

    if faces:
        # Face visible -> Reset covered frames counter
        covered_frames = 0
        face = faces[0]

        eye_dist, _ = face_detector.findDistance(face[LEFT_EYE_TOP], face[LEFT_EYE_BOTTOM])
        face_dist, _ = face_detector.findDistance(face[FACE_LEFT], face[FACE_RIGHT])

        # Eye Aspect Ratio calculation
        ratio = (eye_dist / face_dist) * 100

        if ratio < 11.0:
            closed_frames += 1
        else:
            closed_frames = 0

        if closed_frames >= SLEEP_THRESHOLD_FRAMES:
            is_sleepy = True

        cvzone.putTextRect(img, f"Eye Ratio: {int(ratio)}", (30, 40), scale=1, thickness=1)

    else:
        # Face missing/covered -> Increment covered frames counter
        closed_frames = 0
        covered_frames += 1
        
        if covered_frames >= COVER_THRESHOLD_FRAMES:
            is_face_covered = True

    # ------------------- 2. PHONE DETECTION -------------------
    results = phone_detector.predict(img, stream=True, verbose=False)
    phone_detected = False

    for r in results:
        boxes = r.boxes
        for box in boxes:
            cls_id = int(box.cls[0])
            conf = float(box.conf[0])

            if classNames[cls_id] == "cell phone" and conf > 0.5:
                phone_detected = True
                
                # Bounding box visual
                x1, y1, x2, y2 = map(int, box.xyxy[0])
                cv2.rectangle(img, (x1, y1), (x2, y2), (255, 0, 255), 2)
                cvzone.putTextRect(img, f"Phone detected! {int(conf*100)}%", (x1, max(y1 - 10, 30)), scale=1, thickness=1, colorR=(255, 0, 255))

    # ------------------- 3. ALARM LOGIC & DISPLAY -------------------
    
    # Priority 1: Face covered warning
    if is_face_covered:
        cvzone.putTextRect(img, "DONT COVER YOUR FACE!", (50, 100), scale=2, thickness=3, colorR=(0, 0, 255))
        if not is_audio_busy:
            alarm_facehide.play(0)  # Play full track once
            current_playing = 'facehide'

    # Priority 2: Sleep warning
    elif is_sleepy:
        cvzone.putTextRect(img, "WAKE UP & STUDY!", (50, 100), scale=2, thickness=3, colorR=(0, 0, 255))
        if not is_audio_busy:
            alarm_sleep.play(0)  # Play full track once
            current_playing = 'sleep'

    # Priority 3: Phone warning
    elif phone_detected:
        cvzone.putTextRect(img, "PUT THE PHONE AWAY!", (50, 100), scale=2, thickness=3, colorR=(0, 165, 255))
        if not is_audio_busy:
            alarm_phone.play(0)  # Play full track once
            current_playing = 'phone'

    # Display active warning overlay if audio is still finishing up
    elif is_audio_busy:
        if current_playing == 'facehide':
            cvzone.putTextRect(img, "DONT COVER YOUR FACE!", (50, 100), scale=2, thickness=3, colorR=(0, 0, 255))
        elif current_playing == 'sleep':
            cvzone.putTextRect(img, "WAKE UP & STUDY!", (50, 100), scale=2, thickness=3, colorR=(0, 0, 255))
        elif current_playing == 'phone':
            cvzone.putTextRect(img, "PUT THE PHONE AWAY!", (50, 100), scale=2, thickness=3, colorR=(0, 165, 255))

    cv2.imshow("Smart Study Monitor", img)
    if cv2.waitKey(1) & 0xFF == ord('q'):
        break

cap.release()
cv2.destroyAllWindows()
